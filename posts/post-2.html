看算法很简单，就是随机数生成一个seed，然后seed经过一个算法异或移位啥的求出key，key发到容器获得flag。再看uds的协议，得知27是安全访问，01申请seed，02发key。所以写一个交互脚本

import socket
import time

def calculate_key(seed):
    key = seed ^ 0xA5A5A5A5
    key = ((key << 3) | (key >> 29)) & 0xFFFFFFFF
    key = (key + 0x12345678) & 0xFFFFFFFF
    return key

import hashlib
import itertools

def solve():
    part1 = [60, 54, 49, 58, 51, 61, 59, 50, 54, 49, 24, 54, 50]
    part2 = [47, 25, 47, 56, 55, 54, 48, 57, 24, 57, 47, 24, 24, 25, 25, 62]
    targets = part1 + part2
    target_md5 = "3a22c098710019b31c328a861429d3ad"
    
    possible_chars = []
    for t in targets:
        c1 = t * 2
        c2 = t * 2 + 1
        valid_chars = []
        if 32 <= c1 <= 126:
            valid_chars.append(chr(c1))
        if 32 <= c2 <= 126:
            valid_chars.append(chr(c2))
        possible_chars.append(valid_chars)
    
    count = 0
    for combo in itertools.product(*possible_chars):
        flag = "".join(combo)
        h = hashlib.md5(flag.encode()).hexdigest()
        if h == target_md5:
            print("find flag: " + flag)
            return
        count += 1
        if count % 10000000 == 0:
            print("counting...", count)

if __name__ == '__main__':
    solve()

def dump_payload():

    key = b"hulua"

import struct

def read_dwords(data, offset, count):
    return list(struct.unpack(f'<{count}I', data[offset:offset+count*4]))

def read_bytes(data, offset, count):
    return list(data[offset:offset+count])

def decrypt_sm4(data, base_addr):
    SBOX = read_dwords(data, 0x41ec30 - base_addr, 256)
    FK_CONST = read_dwords(data, 0x422010 - base_addr, 4)
    CK_CONST = read_dwords(data, 0x41f030 - base_addr, 32)
    CT = read_dwords(data, 0x421018 - base_addr, 12)
    print(f"\n[+] 成功提取 SBox (前4个): {SBOX[:4]} ...")
    print(f"[+] 成功提取 FK: {[hex(x) for x in FK_CONST]}")
    print(f"[+] 成功提取 目标密文 CT: {[hex(x) for x in CT]}")

    def L_prime(b):
        return b ^ ((b << 13) | (b >> 19)) & 0xFFFFFFFF ^ ((b << 23) | (b >> 9)) & 0xFFFFFFFF

    def L(b):
        return b ^ ((b << 2) | (b >> 30)) & 0xFFFFFFFF ^ ((b << 10) | (b >> 22)) & 0xFFFFFFFF ^ \
               ((b << 18) | (b >> 14)) & 0xFFFFFFFF ^ ((b << 24) | (b >> 8)) & 0xFFFFFFFF

    def T_prime(val):
        b0, b1, b2, b3 = SBOX[(val >> 24) & 0xFF], SBOX[(val >> 16) & 0xFF], SBOX[(val >> 8) & 0xFF], SBOX[val & 0xFF]
        return L_prime((b0 << 24) | (b1 << 16) | (b2 << 8) | b3)

    def T(val):
        b0, b1, b2, b3 = SBOX[(val >> 24) & 0xFF], SBOX[(val >> 16) & 0xFF], SBOX[(val >> 8) & 0xFF], SBOX[val & 0xFF]
        return L((b0 << 24) | (b1 << 16) | (b2 << 8) | b3)

    def expand_key(key_bytes):
        K = list(struct.unpack('>4I', key_bytes))
        rk = []
        for i in range(4):
            K[i] ^= FK_CONST[i]
        for i in range(32):
            val = T_prime(K[(i+1)%4] ^ K[(i+2)%4] ^ K[(i+3)%4] ^ CK_CONST[i])
            K[i%4] ^= val
            rk.append(K[i%4])
        return rk

    def decrypt_block(ct, rk):
        X = list(ct)
        for i in range(32):
            val = T(X[(i+1)%4] ^ X[(i+2)%4] ^ X[(i+3)%4] ^ rk[31-i])
            X[i%4] ^= val
        return struct.pack('>4I', X[3], X[2], X[1], X[0])

    print("\n[+] 正在根据 Custom SM4 定义进行解密运算...")
    rk = expand_key(b'We1c0me_t0_xmctf')
    flag = b''
    for i in range(0, len(CT), 4):
        flag += decrypt_block(CT[i:i+4], rk)
    print("\n" + "=" * 50)
    print(f"解出的 Flag: {flag.decode('utf-8', errors='ignore')}")
    print("=" * 50)
